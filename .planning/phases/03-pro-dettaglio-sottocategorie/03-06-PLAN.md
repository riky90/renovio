```ts
// wire: numerici come stringhe → normalizzare a number
VoceOut { codice: string | null; descrizione: string; um: string; quantita_stimata: number; prezzo_min: number; prezzo_max: number }
SottocategoriaOut { codice: string | null; nome: string; famiglia: "gerarchica" | "catalogo"; voci: VoceOut[] }
CategoriaDettagliataOut { code: string; nome: string; sottocategorie: SottocategoriaOut[] }
RisposteDomande { bagni: number; porte: number; condizionatori: number }
DettaglioProOut { categorie: CategoriaDettagliataOut[]; risposte: RisposteDomande }
```
